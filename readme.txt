// database/migrations/xxxx_create_users_table.php
public function up(): void
{
    Schema::create('users', function (Blueprint $table) {
        $table->uuid('id')->primary();
        $table->string('phone', 20)->unique();
        $table->string('name', 100)->nullable();

**Why UUID instead of auto-increment:**
UUIDs are not guessable. If you use ID=1, ID=2, ID=3 � anyone can enumerate your restaurants by changing the URL. UUIDs prevent that entirely.

// app/Models/User.php
<?php

namespace App\Models;

use Illuminate\Foundation\Auth\User as Authenticatable;
use Illuminate\Database\Eloquent\Concerns\HasUuids;
use Laravel\Sanctum\HasApiTokens;

class User extends Authenticatable
{
    use HasUuids, HasApiTokens;

    protected $fillable = [
        'phone', 'name', 'role', 'is_verified'
    ];

    protected $hidden = ['remember_token'];

    protected $casts = [
        'is_verified' => 'boolean',
    ];

    // Relationships
    public function restaurant(): HasOne
    {
        return $this->hasOne(Restaurant::class, 'owner_id');
    }

    // Helper methods
    public function isOwner(): bool
    {
        return $this->role === 'owner';
    }

    public function isAdmin(): bool
    {
        return $this->role === 'admin';
    }
}

**Restaurant model:**
```php
// app/Models/Restaurant.php
<?php

namespace App\Models;

use Illuminate\Database\Eloquent\Model;
use Illuminate\Database\Eloquent\Concerns\HasUuids;

class Restaurant extends Model
{
    use HasUuids;

    protected $fillable = [
        'owner_id', 'name', 'description', 'cuisine_types',
        'phone', 'whatsapp', 'address', 'neighborhood', 'city',
        'latitude', 'longitude', 'logo_url', 'cover_photo_url',
        'photos', 'services', 'opening_hours', 'is_open',
        'is_active', 'is_approved'
    ];

    protected $casts = [
        'cuisine_types'  => 'array',
        'photos'         => 'array',
        'services'       => 'array',
        'opening_hours'  => 'array',
        'is_open'        => 'boolean',
        'is_active'      => 'boolean',
        'is_approved'    => 'boolean',
    ];

    // Relationships
    public function owner(): BelongsTo
    {
        return $this->belongsTo(User::class, 'owner_id');
    }

    public function menuCategories(): HasMany
    {
        return $this->hasMany(MenuCategory::class)
                    ->orderBy('display_order');
    }

    public function tables(): HasMany
    {
        return $this->hasMany(RestaurantTable::class)
                    ->where('is_active', true);
    }

    public function profileViews(): HasMany
    {
        return $this->hasMany(ProfileView::class);
    }

    // Accessor: formatted price display
    public function getWhatsappLinkAttribute(): string
    {
        $number = preg_replace('/\D/', '', $this->whatsapp);
        return "https://wa.me/{$number}";
    }

    // Scope: only approved, active restaurants
    public function scopePublic($query)
    {
        return $query->where('is_active', true)
                     ->where('is_approved', true);
    }

    // Scope: filter by open status
    public function scopeOpenNow($query)
    {
        return $query->where('is_open', true);
    }
}
```

**Why `$casts` is critical:**
Without casts, JSON columns come back as plain strings. With `'cuisine_types' => 'array'`, Laravel automatically encodes to JSON on save and decodes to PHP array on read. You never touch `json_encode` or `json_decode` manually.

**SMS Service:**
```php
// app/Services/SmsService.php
<?php

namespace App\Services;

use AfricasTalking\SDK\AfricasTalking;
use Exception;

class SmsService
{
    private $sms;

    public function __construct()
    {
        $AT = new AfricasTalking(
            config('services.africastalking.username'),
            config('services.africastalking.api_key')
        );
        $this->sms = $AT->sms();
    }

    public function sendOtp(string $phone, string $code): bool
    {
        try {
            $this->sms->send([
                'to'      => $this->formatPhone($phone),
                'message' => "RestauHub: Votre code de connexion est {$code}. Valide 10 minutes. Ne partagez pas ce code.",
                'from'    => config('services.africastalking.sender_id'),
            ]);
            return true;
        } catch (Exception $e) {
            // Log but don't crash � in development we'll log to file
            logger()->error('SMS failed: ' . $e->getMessage());
            return false;
        }
    }

    private function formatPhone(string $phone): string
    {
        // Ensure +237 prefix for Cameroon numbers
        $digits = preg_replace('/\D/', '', $phone);
        if (strlen($digits) === 9) {
            return '+237' . $digits;
        }
        if (strlen($digits) === 12 && str_starts_with($digits, '237')) {
            return '+' . $digits;
        }
        return $phone;
    }
}
```

<?php

namespace App\Http\Controllers\Auth;

use App\Http\Controllers\Controller;
use App\Http\Requests\Auth\RequestOtpRequest;
use App\Http\Requests\Auth\VerifyOtpRequest;
use App\Models\OtpCode;
use App\Models\User;
use App\Services\SmsService;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\Auth;
use Illuminate\Support\Str;

class AuthController extends Controller
{
    public function __construct(private SmsService $smsService) {}

    // Show phone entry screen
    public function showLogin()
    {
        return view('auth.login');
    }

    // Step 1: Generate and send OTP
    public function requestOtp(RequestOtpRequest $request)
    {
        $phone = $this->normalizePhone($request->phone);

        // Rate limiting: max 3 OTPs per phone per 10 minutes
        $recentCount = OtpCode::where('phone', $phone)
            ->where('created_at', '>=', now()->subMinutes(10))
            ->count();

        if ($recentCount >= 3) {
            return back()->withErrors([
                'phone' => 'Trop de tentatives. Attendez 10 minutes.'
            ]);
        }

        // Generate 6-digit OTP
        $code = str_pad(random_int(0, 999999), 6, '0', STR_PAD_LEFT);

        // Store OTP with 10 minute expiry
        OtpCode::create([
            'phone'      => $phone,
            'code'       => $code,
            'expires_at' => now()->addMinutes(10),
        ]);

        // Send SMS
        $sent = $this->smsService->sendOtp($phone, $code);

        // In local dev, also log the code so you don't need SMS
        if (app()->isLocal()) {
            logger("OTP for {$phone}: {$code}");
        }

        // Store phone in session to pass to verification screen
        session(['otp_phone' => $phone, 'otp_role' => $request->role ?? 'customer']);

        return redirect()->route('auth.verify');
    }

    // Show OTP entry screen
    public function showVerify()
    {
        if (!session('otp_phone')) {
            return redirect()->route('auth.login');
        }
        return view('auth.verify');
    }

    // Step 2: Verify OTP and log in
    public function verifyOtp(VerifyOtpRequest $request)
    {
        $phone = session('otp_phone');
        $role  = session('otp_role', 'customer');

        if (!$phone) {
            return redirect()->route('auth.login')
                ->withErrors(['otp' => 'Session expir�e. Recommencez.']);
        }

        // Find valid unused OTP
        $otp = OtpCode::where('phone', $phone)
            ->where('code', $request->code)
            ->where('expires_at', '>', now())
            ->where('is_used', false)
            ->latest()
            ->first();

        if (!$otp) {
            return back()->withErrors([
                'code' => 'Code incorrect ou expir�.'
            ]);
        }

        // Mark OTP as used
        $otp->update(['is_used' => true]);

        // Find or create user
        $user = User::firstOrCreate(
            ['phone' => $phone],
            [
                'role'        => $role,
                'is_verified' => true,
            ]
        );

        // Mark verified if first login
        if (!$user->is_verified) {
            $user->update(['is_verified' => true]);
        }

        // Log in with Laravel session
        Auth::login($user, remember: true);

        // Clear OTP session data
        session()->forget(['otp_phone', 'otp_role']);

        // Redirect based on role
        return match($user->role) {
            'owner' => redirect()->route('owner.dashboard'),
            'admin' => redirect()->route('admin.dashboard'),
            default => redirect()->route('home'),
        };
    }

    public function logout(Request $request)
    {
        Auth::logout();
        $request->session()->invalidate();
        $request->session()->regenerateToken();
        return redirect()->route('home');
    }

    private function normalizePhone(string $phone): string
    {
        $digits = preg_replace('/\D/', '', $phone);
        if (strlen($digits) === 9) return '+237' . $digits;
        if (strlen($digits) === 12) return '+' . $digits;
        return $phone;
    }
}

class RestaurantController extends Controller
{
    public function index(Request $request)
    {
        $query = Restaurant::public()
            ->with(['menuCategories' => fn($q) => $q->limit(1)]);

        // Text search
        if ($search = $request->search) {
            $query->where(function($q) use ($search) {
                $q->where('name', 'LIKE', "%{$search}%")
                  ->orWhere('description', 'LIKE', "%{$search}%")
                  ->orWhere('neighborhood', 'LIKE', "%{$search}%");
            });
        }

        // Filter by neighborhood
        if ($neighborhood = $request->neighborhood) {
            $query->where('neighborhood', $neighborhood);
        }

        // Filter open now
        if ($request->boolean('open_now')) {
            $query->openNow();
        }

        // Order by distance if coordinates provided
        if ($request->lat && $request->lng) {
            $query->selectRaw("*,
                (6371 * acos(
                    cos(radians(?)) * cos(radians(latitude)) *
                    cos(radians(longitude) - radians(?)) +
                    sin(radians(?)) * sin(radians(latitude))
                )) AS distance",
                [$request->lat, $request->lng, $request->lat]
            )->orderBy('distance');
        } else {
            $query->latest();
        }

        $restaurants = $query->paginate(20);

        return view('customer.restaurants.index', compact('restaurants'));
    }

    public function show(Restaurant $restaurant)
    {
        // Only show approved, active restaurants publicly
        abort_unless($restaurant->is_active && $restaurant->is_approved, 404);

        // Track profile view (non-blocking)
        $this->trackView($restaurant);

        $menu = $restaurant->menuCategories()
            ->with('items')
            ->get();

        return view('customer.restaurants.show', compact('restaurant', 'menu'));
    }

    private function trackView(Restaurant $restaurant): void
    {
        // Don't count the owner's own views
        if (Auth::id() === $restaurant->owner_id) return;

        // Simple duplicate prevention: one view per IP per hour
        $recentView = $restaurant->profileViews()
            ->where('viewer_ip', request()->ip())
            ->where('created_at', '>=', now()->subHour())
            ->exists();

        if (!$recentView) {
            $restaurant->profileViews()->create([
                'viewer_ip' => request()->ip()
            ]);
            $restaurant->increment('profile_views');
        }
    }
}

class DailyMenuService

    // Get or create today's "Menu du Jour" category
    public function getDailyCategory(Restaurant $restaurant): MenuCategory

        return $restaurant->menuCategories()->firstOrCreate(
            [
                'is_daily_special' => true,
                'name'             => 'Menu du Jour',
            ],
            [
                'display_order' => 0,  // always first
                'name_en'       => "Today's Menu",
            ]

    public function clearDailyMenu(Restaurant $restaurant): void
    {
        $category = $restaurant->menuCategories()
            ->where('is_daily_special', true)
            ->first();

        if ($category) {
            $category->items()->delete();
        }
    }

    // Copy yesterday's daily menu items to today
    public function carryOver(Restaurant $restaurant): int
    {
        $yesterday = Carbon::yesterday();

        // Find items created yesterday in daily special categories
        $yesterdayItems = MenuItem::whereHas('category', fn($q) =>
            $q->where('restaurant_id', $restaurant->id)
              ->where('is_daily_special', true)
        )
        ->whereDate('created_at', $yesterday)
        ->get();

        if ($yesterdayItems->isEmpty()) {
            return 0;
        }

        $todayCategory = $this->getDailyCategory($restaurant);

        // Clear today first to avoid duplicates
        $todayCategory->items()->delete();

        // Re-create yesterday's items for today
        foreach ($yesterdayItems as $item) {
            MenuItem::create([
                'category_id'    => $todayCategory->id,
                'restaurant_id'  => $restaurant->id,
                'name'           => $item->name,
                'name_en'        => $item->name_en,
                'description'    => $item->description,
                'price'          => $item->price,
                'photo_url'      => $item->photo_url,
                'display_order'  => $item->display_order,
                'is_available'   => true,
            ]);
        }

        return $yesterdayItems->count();
    }

// app/Services/QrCodeService.php
<?php

namespace App\Services;

use App\Models\Restaurant;
use App\Models\RestaurantTable;
use Illuminate\Support\Collection;
use SimpleSoftwareIO\QrCode\Facades\QrCode;
use Barryvdh\DomPDF\Facade\Pdf;

class QrCodeService
{
    public function generateAndUpload(
        Restaurant $restaurant,
        RestaurantTable $table
    ): string {
        // The URL the QR encodes
        $menuUrl = route('menu.show', [
            'restaurant' => $restaurant->id,
            'table'      => $table->id,
        ]);

        // Generate QR as SVG (scalable, no pixelation when printed)
        $qrSvg = QrCode::format('svg')
            ->size(400)
            ->errorCorrection('H')  // 30% damage tolerance
            ->margin(2)
            ->generate($menuUrl);

        // Upload SVG to Cloudinary
        $tempPath = tempnam(sys_get_temp_dir(), 'qr_') . '.svg';
        file_put_contents($tempPath, $qrSvg);

        $result = app(CloudinaryService::class)
            ->uploadRaw($tempPath, "restau-hub/qr-codes/{$restaurant->id}");

        unlink($tempPath);

        return $result;
    }

    public function generatePdf(Restaurant $restaurant, Collection $tables)
    {
        // Build QR data URIs for embedding in PDF
        $tableData = $tables->map(function($table) use ($restaurant) {
            $menuUrl = route('menu.show', [
                'restaurant' => $restaurant->id,
                'table'      => $table->id,
            ]);

            $qrBase64 = base64_encode(
                QrCode::format('png')
                    ->size(300)
                    ->errorCorrection('H')
                    ->generate($menuUrl)
            );

            return [
                'table'     => $table,
                'qr_data'   => "data:image/png;base64,{$qrBase64}",
                'menu_url'  => $menuUrl,
            ];
        });

        return Pdf::loadView('owner.tables.qr-pdf', [
            'restaurant' => $restaurant,
            'tables'     => $tableData,
        ])->setPaper('a4');
    }
}

class MenuController extends Controller
{
    public function show(Request $request, Restaurant $restaurant)
    {
        // Must be public and approved
        abort_unless(
            $restaurant->is_active && $restaurant->is_approved,
            404,
            'Restaurant non disponible.'
        );

        // Find the table if provided
        $table = null;
        if ($request->table) {
            $table = $restaurant->tables()
                ->where('id', $request->table)
                ->first();
        }

        // Load full menu with items
        $categories = $restaurant->menuCategories()
            ->with(['items' => fn($q) => $q->orderBy('display_order')])
            ->orderBy('display_order')
            ->get();

        // Move daily special to front if exists
        $categories = $categories->sortByDesc('is_daily_special')->values();

        return view('customer.menu', compact('restaurant', 'categories', 'table'));
    }
}

    public function index()
    {
        $user = Auth::user();

        // If no restaurant yet, redirect to creation
        if (!$restaurant) {

        }

        // Stats for the dashboard

                ->whereDate('created_at', today())

            'views_week'     => $restaurant->profileViews()
                ->where('created_at', '>=', now()->subDays(7))

            'views_by_day'   => $restaurant->profileViews()
                ->where('created_at', '>=', now()->subDays(7))

                ->groupBy('date')
                ->orderBy('date')
                ->get(),

            'menu_items'     => $restaurant->menuCategories()
                ->withCount('items')
                ->get(),

            'total_items'    => MenuItem::where('restaurant_id', $restaurant->id)
                ->count(),

            'available_items' => MenuItem::where('restaurant_id', $restaurant->id)

class OtpAuthTest extends TestCase
{
    use RefreshDatabase;

    public function test_owner_can_request_otp(): void
    {
        $response = $this->post(route('auth.request-otp'), [
            'phone' => '677123456',
            'role'  => 'owner',
        ]);

        $response->assertRedirect(route('auth.verify'));
        $this->assertDatabaseHas('otp_codes', [
            'phone'   => '+237677123456',
            'is_used' => false,
        ]);
    }

    public function test_otp_expires_after_10_minutes(): void
    {
        $otp = OtpCode::create([
            'phone'      => '+237677123456',
            'code'       => '123456',
            'expires_at' => now()->subMinute(), // already expired
        ]);

        session(['otp_phone' => '+237677123456']);

        $response = $this->post(route('auth.verify-otp'), [
            'code' => '123456'
        ]);

        $response->assertSessionHasErrors('code');
    }

    public function test_owner_cannot_access_another_owners_restaurant(): void
    {
        $owner1 = User::factory()->create(['role' => 'owner']);
        $owner2 = User::factory()->create(['role' => 'owner']);
        $restaurant = Restaurant::factory()->create(['owner_id' => $owner1->id]);

        $response = $this->actingAs($owner2)
            ->put(route('owner.restaurant.update', $restaurant), [
                'name' => 'Hacked Name',
            ]);

        $response->assertForbidden();
    }

    public function test_menu_page_loads_without_authentication(): void

            ->create();

        $response->assertOk();
        $response->assertSee($restaurant->name);

// Cache restaurant menu for 60 seconds
// Heavy queries shouldn't run on every QR scan
public function show(Request $request, Restaurant $restaurant)
{
    $categories = Cache::remember(
        "menu:{$restaurant->id}",
        60,  // seconds
        fn() => $restaurant->menuCategories()
            ->with(['items' => fn($q) => $q->orderBy('display_order')])
            ->orderBy('display_order')
            ->get()
    );

    // Clear cache when menu is updated
    // In MenuItemController after any change:
    Cache::forget("menu:{$restaurant->id}");
}

// Security headers middleware
class SecurityHeaders
{
    public function handle(Request $request, Closure $next): Response
    {
        $response = $next($request);
        $response->headers->set('X-Content-Type-Options', 'nosniff');
        $response->headers->set('X-Frame-Options', 'DENY');
        $response->headers->set('Referrer-Policy', 'strict-origin-when-cross-origin');
        return $response;
    }
}
